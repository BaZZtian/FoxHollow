cmake_minimum_required(VERSION 3.25)
project(foxhollow LANGUAGES C CXX)

add_library(gc_disc STATIC runtime/gc_disc.c)
target_include_directories(gc_disc PUBLIC runtime)
target_compile_features(gc_disc PUBLIC c_std_11)

if(UNIX)
  add_library(foxhollow_media STATIC runtime/src/media/gc_media_layer.cpp)
  target_link_libraries(foxhollow_media PUBLIC gc_disc)
  target_include_directories(foxhollow_media PUBLIC runtime/src/media)
  target_compile_features(foxhollow_media PUBLIC cxx_std_17)
endif()

if(CMAKE_C_COMPILER_ID MATCHES "Clang|GNU")
  target_compile_options(gc_disc PRIVATE -Wall -Wextra -Werror -Wconversion)
endif()
if(TARGET foxhollow_media AND CMAKE_CXX_COMPILER_ID MATCHES "Clang|GNU")
  target_compile_options(foxhollow_media PRIVATE -Wall -Wextra -Werror -Wconversion)
endif()
